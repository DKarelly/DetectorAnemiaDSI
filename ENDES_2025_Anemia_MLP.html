ENDES 2023–2025: anemia infantil en la macrorregión norte

Este notebook consolida, limpia y transforma los módulos ENDES (2023, 2024 y 2025) disponibles en BASE DE DATOS para preparar una matriz de entrada para una red MLP. Recorre las subcarpetas, valida cada variable solicitada contra el diccionario Markdown asociado y lee únicamente esas columnas.

Llaves: antropometría y salud infantil por anio + CASEID + orden del niño (HWIDX/HIDX/IDX95); madre por anio + CASEID; hogar por anio + HHID derivado del primer componente de CASEID. Se verifican cardinalidades para impedir que un merge multiplique registros o cruce identificadores entre distintos años.

Dependencias: pandas, numpy y scikit-learn (pip install pandas numpy scikit-learn).

Importante sobre el objetivo: REC44 aporta hemoglobina infantil cruda (HW53), no ajustada ni categorizada. Se deriva la hemoglobina ajustada y la clase de anemia aquí, usando una corrección convencional por altitud y umbrales de 2024 configurables. Confirme ambos con el protocolo clínico aprobado para el estudio antes de publicar resultados. No se usa hemoglobina en las features ni se exporta como predictor.

def dictionary_has_code(dictionary_text, variable):
    return re.search(
        rf"(?<![A-Z0-9_]){re.escape(variable.upper())}(?![A-Z0-9_])",
        dictionary_text.upper(),
    ) is not None

def load_selected_module(module):
    dictionary_path = dictionary_by_code.get(module)
    if dictionary_path is None:
        raise FileNotFoundError(f"No existe Diccionario - {module}.md")
    dictionary_text = dictionary_path.read_text(encoding="utf-8-sig", errors="replace")
    mapping = SOURCE_MAP[module]
    required_codes = set(mapping)
    absent_dictionary = sorted(v for v in required_codes if not dictionary_has_code(dictionary_text, v))
    if absent_dictionary:
        raise ValueError(f"{module}: variables faltantes en diccionario={absent_dictionary}")
    selected = list(mapping)
    yearly_frames = []
    for year in YEARS:
        csv_path = csv_by_code_year[(module, year)]
        header = pd.read_csv(csv_path, encoding="utf-8-sig", nrows=0).columns.tolist()
        absent_csv = sorted(required_codes - set(header))
        if absent_csv:
            raise ValueError(f"{module} ({year}): variables faltantes en CSV={absent_csv}")
        frame = pd.read_csv(
            csv_path, encoding="utf-8-sig", usecols=selected, dtype=str,
            keep_default_na=False, low_memory=False,
        ).rename(columns=mapping)
        frame.insert(0, "anio", year)
        for key in ("anio", "caseid", "hhid", "hhid_mother"):
            if key in frame:
                frame[key] = frame[key].astype("string").str.strip()
        frame = frame.replace(r"^\s*$", pd.NA, regex=True)
        yearly_frames.append(frame)
    return pd.concat(yearly_frames, ignore_index=True)

def assert_unique(frame, keys, label):
    duplicate_count = int(frame.duplicated(keys, keep=False).sum())
    if duplicate_count:
        examples = frame.loc[frame.duplicated(keys, keep=False), keys].head(5).to_dict("records")
        raise ValueError(
            f"{label} no es único por {keys}: {duplicate_count} filas duplicadas. Ejemplos: {examples}"
        )

def merge_checked(left, right, keys, label):
    assert_unique(right, keys, label)
    before = len(left)
    indicator = f"_merge_{label}"
    merged = left.merge(
        right, how="left", on=keys, validate="many_to_one", indicator=indicator,
    )
    if len(merged) != before:
        raise AssertionError(f"El cruce con {label} multiplicó filas ({before} -> {len(merged)})")
    counts = merged[indicator].value_counts(dropna=False).to_dict()
    print(f"  • Cobertura {label}: {counts}")
    return merged.drop(columns=indicator)

5. División estratificada del dataset

Se separan las observaciones en entrenamiento (80 %) y prueba (20 %) antes de ajustar imputadores, escaladores o codificadores. El conjunto de prueba queda reservado para evaluar el modelo; el preprocesador se ajusta solo con entrenamiento y luego transforma ambos conjuntos, evitando fuga de información. La estratificación por nivel de anemia es una práctica habitual en clasificación para mantener la representación de las clases, como se ilustra en estudios de predicción de anemia con un conjunto de prueba independiente (Kamalzadeh et al., 2025; estudio en adultos, split 70/30; documentación de train_test_split).

Nota sobre el desbalance de clases: al consolidar los años ENDES 2023–2025 para la Macrorregión Norte se incrementa sustancialmente la muestra total y los casos de anemia severa respecto a un solo año; stratify=y conserva la distribución de las cuatro clases clínicas de la forma más cercana posible entre entrenamiento y prueba.

def parse_md_dictionary(path):
    """Lee las tablas de un 'Diccionario - XXX.md' y devuelve {VARIABLE: descripcion, tipo, longitud, rango}."""
    entries = {}
    lines = path.read_text(encoding="utf-8-sig", errors="replace").splitlines()
    for line in lines:
        line = line.strip()
        if not line.startswith("|"):
            continue
        body = line[1:-1] if line.endswith("|") else line[1:]
        cells = [c.strip() for c in body.split("|")]
        if len(cells) < 6:
            continue
        # Una celda puede traer varias variables separadas por <br> (p. ej. "ID1<br>HHID").
        names = [n.strip() for n in cells[1].split("<br>") if n.strip()]
        if not names or not all(re.fullmatch(r"[A-Za-z0-9_]+", n) for n in names):
            continue  # encabezados, separadores y filas de continuacion
        fields = [[p.strip() for p in cells[i].split("<br>") if p.strip()] for i in (2, 3, 4, 5)]
        for position, name in enumerate(names):
            values = []
            for parts in fields:
                if len(parts) == len(names):
                    values.append(parts[position])
                elif len(names) == 1:
                    values.append(" ".join(parts))  # texto partido en varias lineas
                else:
                    values.append("")
            entries.setdefault(
                name.upper(), dict(zip(("descripcion_md", "tipo_md", "longitud_md", "rango_md"), values))
            )
    # Algunos .md (p. ej. RECH0) traen parte de la tabla como texto plano separado por <br>:
    # "11 HV009 Total de personas en el hogar N 2 1:25<br>12 HV010 ...". Solo completa lo que falte.
    flat_row = re.compile(r"\d+\s+([A-Za-z][A-Za-z0-9_]*)\s+(.+?)\s+([NA])\s+(\d+)(?:\s+(.*))?")
    range_start = re.compile(r"-?\d+(?::-?\d+)?(?:,\s*-?\d+(?::-?\d+)?)*(?=\s|$)")
    for line in lines:
        if line.strip().startswith("|") or "<br>" not in line:
            continue
        for segment in line.split("<br>"):
            match = flat_row.fullmatch(segment.strip())
            if not match:
                continue
            name, description, kind, length, rest = match.groups()
            found = range_start.match(rest or "")
            entries.setdefault(name.upper(), {
                "descripcion_md": description, "tipo_md": kind, "longitud_md": length,
                "rango_md": found.group(0) if found else "",
            })
    return entries
